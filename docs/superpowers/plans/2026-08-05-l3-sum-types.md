# L3 — Sum Types (payload enums) & Full Match Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Enum variants with one typed payload (`enum Res { Ok(i64), Err(str) }`), construction `Ok(42)`, and `match` with bindings + exhaustiveness — closing language level L3.

**Architecture:** New nominal `TYPE_ENUM` in the checker (plain enums stay `i64`); C backend lowers a sum enum to a tagged struct + union with `static inline` constructors; match on a sum enum compares `.tag` and binds the payload from `.u.v_Variant`. LLVM backend gives honest `unsupported`.

**Tech Stack:** C99 compiler (`src/*.c`, `include/baga.h`), Baga test suite (`tests/std/*_test.baga` via `scripts/baga-test`), grep probes in `scripts/run_tests.sh`.

**Spec:** `docs/superpowers/specs/2026-08-05-l3-sum-types-design.md`

## Global Constraints

- Plain enums (no payloads) keep today's semantics byte-for-byte: `i64`, bare variants, C `typedef enum`. Existing tests (`examples/cvet.baga`, `examples/match.baga`, `tests/std/enum_test.baga` if present) must stay green untouched.
- Sum-enum payload is **exactly one** type; multi-field data goes in a struct payload.
- Sum-enum variant names are globally unique across enums AND function names (checker error, post-pass like the L6 duplicate pass at `src/checker.c:1723-1747`).
- `Vec<sum enum>` / `Map<K, sum enum>` stay rejected — reuse the existing "неподдържан елементен тип" errors (this supersedes the custom message in the spec; equally honest).
- Every task ends green: `make -j4 && make test` (and `make self` at the end). LLVM oracle (`make llvm && make test-llvm`) only if the toolchain is present.
- Test-runner idiom: tests print `ok <name>` per check via a local `check(name, ok)` fn (copy from `tests/std/fnval_test.baga:7-14`), end with a summary print; exit code decides pass/fail under `scripts/baga-test`.
- Commits: one per task, message style `feat(lang): ...` (see `git log`).

- [ ] **Step 1: Replace the NODE_MATCH case**

Keep the existing loop for non-enum scrutinees; add the TYPE_ENUM path:
```c
case NODE_MATCH: {
    Type *st = infer(ctx, n->match_expr);
    if (st && st->kind == TYPE_ENUM && st->name) {
        /* L3: match върху sum enum — вариантни патерни, bindings, пълнота */
        Node *ed = NULL;
        for (int ei = 0; ei < ctx->n_enums; ei++)
            if (strcmp(ctx.enums[ei].name, st->name) == 0)
                { ed = ctx.enums[ei].decl; break; }
        if (!ed) { t = type_new(TYPE_ERROR); break; }
        int nv = ed->n_variants;
        int *covered = calloc((size_t)nv, sizeof(int));
        int has_wild = 0;
        t = type_new(TYPE_VOID);
        for (int i = 0; i < n->match_arms.len; i++) {
            Node *arm = n->match_arms.data[i];
            if (!arm->arm_pattern) {
                has_wild = 1;
            } else if (arm->arm_pattern->kind != NODE_IDENT) {
                check_error(ctx, arm->arm_pattern->pos,
                    "match върху '%s': патернът трябва да е вариант или '_'",
                    st->name);
            } else {
                const char *vn = arm->arm_pattern->name;
                int vidx = -1;
                for (int j = 0; j < nv; j++)
                    if (strcmp(ed->enum_variants[j], vn) == 0) { vidx = j; break; }
                if (vidx < 0) {
                    check_error(ctx, arm->arm_pattern->pos,
                        "патернът '%s' не е вариант на enum '%s'", vn, st->name);
                } else {
                    covered[vidx] = 1;
                    int has_pl = ed->enum_payloads && ed->enum_payloads[vidx];
                    if (has_pl && !arm->arm_binding)
                        check_error(ctx, arm->arm_pattern->pos,
                            "вариантът '%s' има payload — нужен е binding: %s(v)",
                            vn, vn);
                    if (!has_pl && arm->arm_binding)
                        check_error(ctx, arm->arm_pattern->pos,
                            "вариантът '%s' няма payload", vn);
                }
            }
            push_scope(ctx);
            if (arm->arm_binding && arm->arm_pattern &&
                arm->arm_pattern->kind == NODE_IDENT) {
                Type *pt = type_new(TYPE_ERROR);
                for (int j = 0; j < nv; j++)
                    if (strcmp(ed->enum_variants[j], arm->arm_pattern->name) == 0 &&
                        ed->enum_payloads && ed->enum_payloads[j])
                        pt = resolve_type_node(ctx, ed->enum_payloads[j]);
                env_define(ctx, arm->arm_binding, pt, arm->pos);
            }
            Type *bt = infer(ctx, arm->arm_body);
            pop_scope(ctx);
            if (bt->kind == TYPE_VOID && arm->arm_body &&
                arm->arm_body->kind == NODE_BLOCK &&
                arm->arm_body->stmts.len > 0) {
                Node *last = arm->arm_body->stmts.data[arm->arm_body->stmts.len - 1];
                if (last->kind == NODE_RETURN && last->ret_val && last->ret_val->type)
                    bt = last->ret_val->type;
            }
            if (i == 0) t = bt;
            else if (!type_eq(bt, t))
                check_error(ctx, arm->pos,
                    "arm #%d на match е от тип %s, но първият arm е %s",
                    i + 1, type_str(bt), type_str(t));
        }
        if (!has_wild)
            for (int j = 0; j < nv; j++)
                if (!covered[j])
                    check_error(ctx, n->pos,
                        "match върху '%s' не е пълен — липсва вариант '%s' (или добави '_')",
                        st->name, ed->enum_variants[j]);
        free(covered);
        break;
    }
    /* не-enum match: досегашното поведение (първият arm дава типа) */
    t = type_new(TYPE_VOID);
    for (int i = 0; i < n->match_arms.len; i++) {
        Node *arm = n->match_arms.data[i];
        if (arm->arm_pattern) infer(ctx, arm->arm_pattern);
        Type *bt = infer(ctx, arm->arm_body);
        if (bt->kind == TYPE_VOID && arm->arm_body &&
            arm->arm_body->kind == NODE_BLOCK &&
            arm->arm_body->stmts.len > 0) {
            Node *last = arm->arm_body->stmts.data[arm->arm_body->stmts.len - 1];
            if (last->kind == NODE_RETURN && last->ret_val && last->ret_val->type)
                bt = last->ret_val->type;
        }
        if (i == 0) t = bt;
    }
    break;
}
```

import "std/str/str.baga"

fn check(name: str, ok: bool) {
    if ok {
        print(concat("ok ", name))
    } else {
        eprintln(concat("FAIL ", name))
        exit(1)
    }
}

enum Res { Ok(i64), Err(str) }
enum Opt { Some(str), None }
struct Point { x: i64, y: i64 }
enum Shape { Dot, Circle(Point) }

fn unwrap_or(r: Res, dflt: i64) -> i64 {
    return match r { Ok(v) => v, _ => dflt }
}

fn describe(o: Opt) -> str {
    return match o {
        Some(s) => s,
        None => "нищо",
    }
}

fn make_ok(x: i64) -> Res {
    return Ok(x)
}

fn main() {
    let r = Ok(42)
    check("construct_match", match r { Ok(v) => v, Err(e) => 0 } == 42)

    let e = Err("гръм")
    check("err_wildcard", unwrap_or(e, -1) == -1)
    check("ok_unwrap", unwrap_or(r, -1) == 42)
    check("fn_return", unwrap_or(make_ok(7), 0) == 7)

    check("none_bare", describe(None) == "нищо")
    check("some_str", describe(Some("има")) == "има")

    let c = Circle(Point { x: 3, y: 4 })
    let area = match c {
        Circle(p) => p.x * p.x + p.y * p.y,
        Dot => 0,
    }
    check("struct_payload", area == 25)
    check("dot_tag", match Dot { Dot => 1, Circle(p) => 0 } == 1)

    let msg = match e { Ok(v) => "ок", Err(m) => m }
    check("err_binding", msg == "гръм")

    let rr = match r {
        Ok(v) => match e { Ok(w) => v + w, Err(m) => v },
        Err(m) => 0,
    }
    check("nested_match", rr == 42)

    let mut cur = Ok(1)
    cur = Err("сменен")
    check("reassign", unwrap_or(cur, 9) == 9)

    let ann: Res = Ok(5)
    check("annotation", unwrap_or(ann, 0) == 5)

    match r {
        Ok(v) => print(concat("stmt_arm ", int_to_str(v))),
        Err(m) => print(m),
    }

    print("sumtype: all passed")
}
